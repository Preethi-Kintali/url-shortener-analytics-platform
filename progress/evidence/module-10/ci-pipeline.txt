name: ci

on:
  pull_request:
  push:
    branches: [main]

jobs:
  nestjs:
    runs-on: ubuntu-latest
    defaults:
      run:
        working-directory: api
    services:
      postgres:
        image: postgres:16-alpine
        env:
          POSTGRES_USER: postgres
          POSTGRES_PASSWORD: postgres
          POSTGRES_DB: upsk_sdf_test
        ports:
          - 5432:5432
        options: >-
          --health-cmd "pg_isready -U postgres -d upsk_sdf_test"
          --health-interval 5s
          --health-timeout 5s
          --health-retries 10
      redis:
        image: redis:7-alpine
        ports:
          - 6379:6379
        options: >-
          --health-cmd "redis-cli ping"
          --health-interval 5s
          --health-timeout 5s
          --health-retries 10

    env:
      DATABASE_URL_TEST: postgresql://postgres:postgres@localhost:5432/upsk_sdf_test?schema=public
      DATABASE_URL: postgresql://postgres:postgres@localhost:5432/upsk_sdf_test?schema=public
      REDIS_URL: redis://localhost:6379
      JWT_SECRET: ci-test-secret-value-must-be-at-least-32-chars-long!
      PORT: 3000

    steps:
      - uses: actions/checkout@v4
      - uses: pnpm/action-setup@v4
        with:
          version: 9
      - uses: actions/setup-node@v4
        with:
          node-version: 20
          cache: pnpm
          cache-dependency-path: api/pnpm-lock.yaml
      - name: Install dependencies
        run: pnpm install --frozen-lockfile
      - name: Generate Prisma Client
        run: npx prisma generate
      - name: Run one-off database migrations
        run: npx prisma migrate deploy
      - name: Run integration tests
        run: pnpm run test
      - name: Build production bundle
        run: pnpm run build
